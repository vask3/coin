ESP32 Hardware & IoT Firmware Guides

Изберете софтуер за вашата ESP32 платка. Заменете YOUR_WIFI_SSID, YOUR_USERNAME и вашия MINING_KEY от таблото.

1. ESP32 Standard System Node (Без сензори)

Следи вътрешните системни параметри на чипа (Free Heap, Uptime).

#include <WiFi.h>
#include <HTTPClient.h>
#include <sha256.h>

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";
const char* serverSubmitUrl = "https://coin.vasko.blog/submit_block";

String username = "YOUR_USERNAME";
String miningKey = "YOUR_MINING_KEY";

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    unsigned long nonce = random(0, 2147483647);
    String targetData = "ESP32_SYS_HEAP:" + String(ESP.getFreeHeap());
    String combined = targetData + String(nonce);

    Sha256.init();
    Sha256.print(combined.c_str());
    byte* hash = Sha256.result();
    String hashStr = "";
    for (int i = 0; i < 32; i++) { char buf[3]; sprintf(buf, "%02x", hash[i]); hashStr += buf; }

    if (hashStr.startsWith("0000")) {
      HTTPClient http;
      http.begin(serverSubmitUrl);
      http.addHeader("Content-Type", "application/json");
      String jsonBody = "{\"username\":\"" + username + "\",\"mining_key\":\"" + miningKey + "\",\"nonce\":\"" + String(nonce) + "\",\"target\":\"" + targetData + "\"}";
      http.POST(jsonBody);
      http.end();
      delay(5000);
    }
  }
}

2. ESP32 Climate Node (Температура и Влажност с DHT22)

Свързване: DHT22 Data пин към GPIO 4

#include <WiFi.h>
#include <HTTPClient.h>
#include <sha256.h>
#include <DHT.h>

#define DHTPIN 4
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";
const char* serverSubmitUrl = "https://coin.vasko.blog/submit_block";

String username = "YOUR_USERNAME";
String miningKey = "YOUR_MINING_KEY";

void setup() {
  Serial.begin(115200);
  dht.begin();
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); }
}

void loop() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  if (isnan(h) || isnan(t)) { delay(2000); return; }

  unsigned long nonce = random(0, 2147483647);
  String targetData = "TEMP:" + String(t) + "_HUM:" + String(h);
  String combined = targetData + String(nonce);

  Sha256.init();
  Sha256.print(combined.c_str());
  byte* hash = Sha256.result();
  String hashStr = "";
  for (int i = 0; i < 32; i++) { char buf[3]; sprintf(buf, "%02x", hash[i]); hashStr += buf; }

  if (hashStr.startsWith("0000")) {
    HTTPClient http;
    http.begin(serverSubmitUrl);
    http.addHeader("Content-Type", "application/json");
    String jsonBody = "{\"username\":\"" + username + "\",\"mining_key\":\"" + miningKey + "\",\"nonce\":\"" + String(nonce) + "\",\"target\":\"" + targetData + "\"}";
    http.POST(jsonBody);
    http.end();
    delay(5000);
  }
}
